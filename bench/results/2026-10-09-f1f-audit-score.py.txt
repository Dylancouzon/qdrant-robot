"""Lead audit and shared three-vote judging of the completed paired Jetson answers."""
import hashlib
import json
import statistics
import numpy as np
from pathlib import Path
from eval.judge import MODEL, VOTES, score_items

BASE = Path('bench/results')
RAW = BASE / '2026-10-09-paired-event-answers-1791549264663591169.jsonl'
def sha(p): return hashlib.sha256(p.read_bytes()).hexdigest()
rows = [json.loads(line) for line in RAW.read_text().splitlines()]
assert rows[-1] == {'summary':'paired_event_answers','completed':True,'n':64,'native_exit':0}
assert rows[-2]['type'] == 'native_exit' and rows[-2]['native_exit'] == 0
assert rows[0]['probe_sha256'] == sha(BASE/'2026-10-09-f1f-paired-answers.py.txt')
for path, expected in rows[1]['implementation_hashes'].items():
    assert sha(Path(path)) == expected, path
cases = [json.loads(line) for line in Path('eval/dev/event-retrieval-2.jsonl').read_text().splitlines()]
assert sha(Path('eval/dev/event-retrieval-2.jsonl')) == rows[1]['fixture_sha256']
cases = {case['id']:case for case in cases}
answers = [row for row in rows if row.get('type') == 'answer']
assert len(answers) == 64 and len({row['id'] for row in answers}) == 64
for arm in ('production','calibrated'):
    assert {row['step'] for row in answers if row['arm']==arm} == set(cases)
for row in answers:
    case=cases[row['step']]
    assert row['heard']==case['question'] and row['rubric']==case['rubric']
    assert row['budget']['prompt_token_bound']+64<=2048
    assert row['answer'].strip() and row['first_audio_s']>0
    for event in row['delivered']:
        source=case['sources'][event['source_id']-1]
        assert row['source_ids'][str(event['source_id'])]==source['id']
        text=source['caption' if event['kind']=='observation' else 'said']
        assert event['source_hash']==hashlib.sha256(
            f"{event['source_id']}\0{event['kind']}\0{text}".encode()).hexdigest()
        for span in event['spans']:
            assert span['text']==text[span['start']:span['end']]
    decisive=case['decisive_spans']
    delivered=sum(any(row['source_ids'][str(e['source_id'])]==g['source_id']
                     and e['kind']==g['kind'] and s['start']<=g['start'] and s['end']>=g['end']
                     and s['text'][g['start']-s['start']:g['end']-s['start']]==g['text']
                     for e in row['delivered'] for s in e['spans']) for g in decisive)
    assert delivered==row['metrics']['decisive_spans_delivered']
    if row['arm']=='calibrated':assert delivered==len(decisive)
items=[{'id':r['id'],'visitor':r['heard'],'robot':r['answer'],'rubric':r['rubric']} for r in answers]
print('Native completion/source/span/budget audit passed for64 answers.',flush=True)
verdicts, fresh = score_items(items)
judged=RAW.with_suffix('.judged.jsonl')
judged.write_text(''.join(json.dumps({**item,**verdict})+'\n'
                         for item,verdict in zip(items,verdicts,strict=True)))
by_id={v['id']:v for v in verdicts}
summary={}
for arm in ('production','calibrated'):
    selected=[r for r in answers if r['arm']==arm]
    warm=[r for r in selected if not r['cold']]
    summary[arm]={'correct':sum(by_id[r['id']]['pass'] for r in selected),'n':len(selected),
        'warm_n':len(warm),'warm_first_audio_p50_s':statistics.median(r['first_audio_s'] for r in warm),
        'warm_first_audio_p95_s': float(np.percentile([r['first_audio_s'] for r in warm],95)),
        'cold_first_audio_s':next(r['first_audio_s'] for r in selected if r['cold']),
        'budget_p50_s':statistics.median(r['budget_s'] for r in warm),
        'facts_delivered':sum(r['metrics']['decisive_spans_delivered'] for r in selected),
        'unrelated_returns':sum(r['metrics']['unrelated_returns'] for r in selected),
        'dropped':sum(len(r['dropped']) for r in selected),'categories':{
            cat:{'correct':sum(by_id[r['id']]['pass'] for r in selected if r['category']==cat),
                 'n':sum(r['category']==cat for r in selected)} for cat in sorted({r['category'] for r in selected})}}
pairs={r['step']:{a:next(x for x in answers if x['step']==r['step'] and x['arm']==a)
                  for a in ('production','calibrated')} for r in answers}
wins=sum(not by_id[p['production']['id']]['pass'] and by_id[p['calibrated']['id']]['pass'] for p in pairs.values())
losses=sum(by_id[p['production']['id']]['pass'] and not by_id[p['calibrated']['id']]['pass'] for p in pairs.values())
telemetry=[json.loads(line) for line in RAW.with_suffix('.telemetry.jsonl').read_text().splitlines()]
loaded=[r for r in telemetry if r['phase'] in ('paired','answer')]
result={'type':'paired_answer_audit','measurement':'J controlled source-to-answer; no ASR input/history/image',
    'result':RAW.name,'result_sha256':sha(RAW),'judged_sha256':sha(judged),
    'audit_sha256':sha(Path(__file__)),'completed':True,'native_exit':0,'judge':MODEL,'votes':VOTES,
    'fresh_judgments':fresh,'summary':summary,'paired_wins':wins,'paired_losses':losses,
    'warm_p50_regression_s':summary['calibrated']['warm_first_audio_p50_s']-summary['production']['warm_first_audio_p50_s'],
    'minimum_loaded_sampled_headroom_mib':min(r['resources']['mem_available_kb'] for r in loaded)/1024,
    'timed_arm_paging':{arm:{key:sum(r['paging_delta'][key] for r in answers if r['arm']==arm)
                          for key in ('pswpin','pswpout')} for arm in ('production','calibrated')},
    'minimum_turn_detect_fps':min(r['detect_fps'] for r in answers),
    'p95_method':'numpy linear percentile on31 warm samples',
    'limits':'Not physical/end-of-speech latency, full-turn integration, representative-volume cost, or sustained acceptance.'}
(BASE/'2026-10-09-f1f-operations.jsonl').write_text(json.dumps(result)+'\n')
print(json.dumps(result,indent=2),flush=True)
