import torch
from modelopt.torch.quantization.extensions import get_cuda_ext
ext = get_cuda_ext(raise_if_failed=True)
if ext is None:
    raise RuntimeError("Official CUDA extension did not load")
print("EXTENSION", ext.__file__, flush=True)
x = torch.tensor([[-1., -.5, 0., .5, 1.]], device="cuda", dtype=torch.float16)
a = torch.tensor([1.], device="cuda", dtype=torch.float32)
y = ext.fake_tensor_quant_with_axis(x, a, 0, 4, False, True)
expected = ((x.float() * 7).round().clamp(-7, 7) / 7).half()
torch.testing.assert_close(y, expected, rtol=0, atol=0)
torch.cuda.synchronize()
print("OFFICIAL_INT4_AXIS_CUDA_CHECK_PASS", flush=True)
