"""Launch unchanged continuous smoke with an explicitly bound worker observer."""
import hashlib
import os
import sys
from pathlib import Path

root = '/home/qdrant/qdrant-robot/releases/038099d89da2'
os.chdir(root)
sys.path.insert(0, root)
os.environ['ROBOT_RELEASE'] = '038099d89da2'
from eval import continuous

WORKER = '"""Time actual continuous scheduler and native calls without changing their inputs."""\nimport functools\nimport hashlib\nimport itertools\nimport json\nimport os\nimport sys\nimport threading\nimport time\nfrom pathlib import Path\n\nroot, timing = sys.argv[1:3]\nos.chdir(root)\nsys.path.insert(0, root)\nsys.argv = [\'eval.continuous\', *sys.argv[3:]]\nfd = os.open(timing, os.O_WRONLY | os.O_CREAT | os.O_APPEND, 0o644)\nids = itertools.count()\n\n\ndef observe(kind, **fields):\n    row = dict(type=kind, wall_time=time.time(), monotonic=time.monotonic(),\n               thread=threading.get_native_id(), **fields)\n    data = (json.dumps(row) + \'\\n\').encode()\n    assert len(data) < 4096\n    assert os.write(fd, data) == len(data)\n\n\ndef wrap(owner, name, label):\n    original = getattr(owner, name)\n\n    @functools.wraps(original)\n    def measured(*args, **kwargs):\n        call = next(ids)\n        observe(\'call_start\', call=call, name=label)\n        started = time.perf_counter()\n        try:\n            return original(*args, **kwargs)\n        finally:\n            observe(\'call_end\', call=call, name=label,\n                    elapsed_s=time.perf_counter() - started)\n\n    setattr(owner, name, measured)\n\n\nfrom eval import continuous\nfrom robot import asr, brain, camera, detect\n\nfor owner, name, label in (\n    (continuous.Scheduler, \'report\', \'scheduler_report\'),\n    (continuous.Scheduler, \'drain\', \'scheduler_drain\'),\n    (continuous.Scheduler, \'feed\', \'scheduler_feed\'),\n    (continuous, \'resources\', \'evaluator_resources\'),\n    (camera.FileCamera, \'latest\', \'source_latest\'),\n    (asr.Ears, \'feed\', \'ears_feed\'),\n    (asr.Scribe, \'__call__\', \'asr\'),\n    (detect._TensorRT, \'__call__\', \'detector_native\'),\n    (brain.Brain, \'budget\', \'prompt_budget\'),\n    (brain.Brain, \'prefill\', \'prefill\'),\n):\n    wrap(owner, name, label)\nobserve(\'observer_started\', command=sys.argv, release=os.environ[\'ROBOT_RELEASE\'],\n        implementation_hashes={p: hashlib.sha256(Path(p).read_bytes()).hexdigest()\n                               for p in (\'eval/continuous.py\', \'robot/camera.py\',\n                                         \'robot/main.py\', \'robot/brain.py\', \'models.json\')})\nrc = continuous.main()\nobserve(\'observer_finished\', returned_exit=rc)\nos.close(fd)\nsys.exit(rc)\n'
original = continuous.supervise

output = '/home/qdrant/qdrant-robot/shared/results/2026-10-09-f1i-smoke-candidate-a.jsonl'
launcher_sha = sys.argv[-1]
sys.argv = ['eval.continuous',
            'eval/dev/continuous-smoke-20261009/visits.jsonl', '--audio',
            'eval/dev/continuous-smoke-20261009/audio', '--output', output]
# Save the launcher binding before main parses its unchanged CLI.
continuous.emit(Path(output).with_suffix('.launch.jsonl'),
                dict(type='launcher', launcher_sha256=launcher_sha,
                     command=sys.argv, release=os.environ['ROBOT_RELEASE']))
def bound_supervise(command, output, *args, **kwargs):
    executed = [command[0], '-u', '-c', WORKER, root,
                str(output.with_suffix('.timing.jsonl')), *command[3:]]
    assert command[1:3] == ['-m', 'eval.continuous']
    continuous.emit(output.with_suffix('.launch.jsonl'),
                    dict(type='executed_worker', command=executed,
                         worker_sha256=hashlib.sha256(WORKER.encode()).hexdigest(),
                         launcher_sha256=launcher_sha,
                         instrumentation='start/end timing only; logging adds overhead'))
    return original(executed, output, *args, **kwargs)

continuous.supervise = bound_supervise
sys.exit(continuous.main())
